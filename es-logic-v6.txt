// Safe division: returns na instead of inf / 0-divide garbage.
f_div(float nom, float den) =>
    na(den) or den == 0 ? float(na) : nom / den

f_percent(float nom, float den) =>
    100 * nz(f_div(nom, den))

f_roundUp(float number, int decimals) =>
    factor = math.pow(10, decimals)
    math.ceil(number * factor) / factor

// ═══════════════════════════════════════════════════════════════════════════════
//  5.  HIGHER-TIMEFRAME REQUESTS  —  NO-REPAINT
//      v5 called request.security(..., lookahead = barmerge.lookahead_on) on
//      *intraday* expressions with no history offset. On historical bars that
//      returns the FINAL value of an HTF bar that had not closed yet, so every
//      13m / 15m / 10m / 2m signal below was reading the future. Backtests
//      looked perfect and live trading did not match.
//
//      f_htf() uses lookahead_off, so a chart bar only ever sees higher-
//      timeframe data that already existed at that point in time. History is
//      then fixed, and the live bar develops and settles at its close, which
//      is what a live indicator should do.
//      "Strict HTF" additionally drops the forming HTF bar: values never move
//      intrabar, at the cost of one extra HTF bar of lag.
// ═══════════════════════════════════════════════════════════════════════════════
strictHTF = input.bool(false, "Strict HTF (ignore the forming HTF bar)", group = G_HTF, tooltip = "OFF = the live higher-timeframe bar develops during the session. Non-repainting on history.\nON  = only fully closed higher-timeframe bars are used. Nothing moves intrabar, but signals lag by one HTF bar.")
htfOff = strictHTF ? 1 : 0

f_htf(simple string tf, float expr) =>
    request.security(syminfo.tickerid, tf, expr[htfOff], lookahead = barmerge.lookahead_off)

// v5 declared this input inside a ternary branch. Inputs must live at top level.

// Guard: a flat close series makes stdev 0 → division by zero in the weight.
gmaSigmaSafe = math.max(nz(gmaSigma), 1e-10)

float gma = 0.0
float sumOfWeights = 0.0
for i = 0 to gamaLength - 1
    weight = math.exp(-math.pow((i - (gamaLength - 1)) / (2 * gmaSigmaSafe), 2) / 2)
    value  = ta.highest(avpchange, i + 1) + ta.lowest(avpchange, i + 1)
    gma := gma + (value * weight)
    sumOfWeights := sumOfWeights + weight

gma := nz(f_div(gma, sumOfWeights)) / 2
gma := ta.ema(gma, 7)

almaBuy  = ta.crossover(avpchange, gma)  and barstate.isconfirmed
almaSell = ta.crossunder(avpchange, gma) and barstate.isconfirmed

t3c2 = 1 - t3c1
t3c3 =  3.0 * (0.4 * 0.4 + 0.4 * 0.4 * 0.4)
t3c4 = -3.0 * (2.0 * 0.4 * 0.4 + 0.4 + 0.4 * 0.4 * 0.4)
t3c5 =  3.0 * 0.4 + 1.0 + 0.4 * 0.4 * 0.4 + 3.0 * 0.4 * 0.4

var float i2 = na
var float i3 = na
var float i4 = na
var float i5 = na
var float i6 = na

i2 := t3c1 * i1    + t3c2 * nz(i2[1])
i3 := t3c1 * i2    + t3c2 * nz(i3[1])
i4 := t3c1 * i3    + t3c2 * nz(i4[1])
i5 := t3c1 * i4    + t3c2 * nz(i5[1])
i6 := t3c1 * i5    + t3c2 * nz(i6[1])
cto = -0.4 * 0.4 * 0.4 * i6 + t3c3 * i5 + t3c4 * i4 + t3c5 * i3

long  = (close > cto and close[1] < cto[1] and close > close[1]) or (close[1] > cto[1] and close > close[1] and close[1] < close[2] and close > ema3)
short = (close < cto and close[1] > cto[1] and close < close[1]) or (close[1] < cto[1] and close < close[1] and close[1] > close[2] and close < ema3)
profit_long  = close < close[1] and low  < ema3 and close[1] > ema3[1] and close[2] > ema3[2]
profit_short = close > close[1] and high > ema3 and close[1] < ema3[1] and close[2] < ema3[2]

gain2 = math.abs(close[1] - close[2])
gainRatio = f_div(gain1 - gain2, gain2)          // v5 divided by gain2 unguarded
bodyPos   = f_div(close - low, high - low)       // v5 divided by (high-low) unguarded

pattern_call = close[1] < close[2] and close[2] < close[3] and low <= low[1] and close >= close[1] and not na(gainRatio) and gainRatio >= 0.50 and wma11_13 > wma48_13 and not na(bodyPos) and bodyPos > 0.70
// (high - low) - (high - close) == close - low, so v5's second branch is just
// bodyPos < 0.25.
pattern_put  = (close[1] > close[2] and close[2] > close[3] and high <= high[1] and close <= close[1] and not na(gainRatio) and gainRatio <= -0.5) or (not na(bodyPos) and bodyPos < 0.25 and wma11_13 < wma48_13)

if pattern_put
    cnt_put := cnt_put + 1
if pattern_put and cnt_put == 1
    cnt_out_up := cnt_out_up + 1
if bottomsupport

call_100 = pattern_call and long
put_100  = pattern_put and cnt_put == 1 and short

// Big reverse
oc   = math.abs(open[1] - close[1])

            else if h2_val == 0.0 and math.abs(ph - h1_val) > 3
                h2_val := ph
            else if h3_val == 0.0 and h2_val != 0.0 and math.abs(ph - h2_val) > 3 and math.abs(ph - h1_val) > 3
                h3_val := ph
        if not na(pl)
            if l1_val == 0.0
                l1_val := pl
            else if l2_val == 0.0 and math.abs(pl - l1_val) > 3
                l2_val := pl
            else if l3_val == 0.0 and l2_val != 0.0 and math.abs(pl - l2_val) > 3 and math.abs(pl - l1_val) > 3
                l3_val := pl

// v5 bug: the S2 label printed l3_val, so S2 and S3 always showed the same
// number on the chart (visible in the screenshot: "S2: 7,819.25 / S3: 7,819.25").
// Fixed by indexing the levels here.
srVals = array.from(h1_val, h2_val, h3_val, l1_val, l2_val, l3_val)
var srTags = array.from("R1", "R2", "R3", "S1", "S2", "S3")

var array<line>  srLines  = array.new<line>(6, na)
var array<label> srLabels = array.new<label>(6, na)

if barstate.islast and showSR
    for k = 0 to 5
        float lvl = array.get(srVals, k)
        if lvl != 0.0 and not na(lvl)
            string tag = array.get(srTags, k)
            color  col = k < 3 ? color.white : color.blue
            line  ln = array.get(srLines, k)
            label lb = array.get(srLabels, k)
            if na(ln)
                array.set(srLines, k, line.new(bar_index + line_start, lvl, bar_index + line_len, lvl, width = 2, color = col))
            else
                line.set_xy1(ln, bar_index + line_start, lvl)
                line.set_xy2(ln, bar_index + line_len, lvl)
            if na(lb)
                array.set(srLabels, k, label.new(bar_index + line_len2, lvl + 1, tag + ": " + str.tostring(lvl, format.mintick), style = label.style_text_outline, color = color.new(#454B1B, 50), textcolor = color.white))
            else
                label.set_xy(lb, bar_index + line_len2, lvl + 1)
                label.set_text(lb, tag + ": " + str.tostring(lvl, format.mintick))

// Level-cross annotations. These only ever fire on the live bar (h1..l3 are
// computed on barstate.islast only) — same as v5 — but they are now gated on a
// confirmed bar so one cross produces exactly one label instead of one per tick.
float maxCrossUp = 0.0
float maxCrossDn = 0.0

if showSR and barstate.isconfirmed
    for k = 0 to 5
        float lvl = array.get(srVals, k)
        if lvl != 0.0 and not na(lvl)
            string tag = array.get(srTags, k)
            float  pts = math.abs(close - open)
            if (open < lvl or close[1] < lvl) and close > lvl and pts > 0
                maxCrossUp := math.max(maxCrossUp, pts)
                label.new(bar_index + 6, lvl, "U-" + tag + ": " + str.tostring(pts, "#.##"), style = label.style_text_outline, color = color.new(#454B1B, 50), textcolor = color.yellow)
            if (open > lvl or close[1] > lvl) and close < lvl and pts > 0
                maxCrossDn := math.max(maxCrossDn, pts)
                label.new(bar_index + 6, lvl, "D-" + tag + ": " + str.tostring(pts, "#.##"), style = label.style_text_outline, color = color.new(#454B1B, 50), textcolor = color.red)

strongSRCrossUp = maxCrossUp >= cross_up_point
strongSRCrossDn = maxCrossDn >= cross_down_point
